// Realizar Sorteo Oficial con Lógica Circular (Anillo) function realizarSorteoOficial() { const rawLines = document.getElementById('listInput').value.trim().split('\n'); const rawBlacklist = document.getElementById('blacklistInput').value.trim().split('\n'); let participantesValidos = []; rawLines.forEach(line => { let lineaLimpia = line.trim(); if (!lineaLimpia) return; const matchUser = lineaLimpia.match(/@[\w_]+/); if (matchUser) { let boletoLimpio = lineaLimpia.replace(/PUNTO\s+EXTRA\s*/gi, '').trim(); participantesValidos.push({ ticket: boletoLimpio, user: matchUser[0] }); } }); const total = participantesValidos.length; if (total === 0) { alert("No se han encontrado participantes válidos con el formato 'número-@usuario'."); return; } const blacklist = rawBlacklist .map(u => u.trim().toLowerCase()) .filter(u => u.length > 0); let log = []; let ganadorEncontrado = null; let indexGanadorEnListaOriginal = -1; // Selección aleatoria let listaCopia = [...participantesValidos]; while (listaCopia.length > 0) { const indexAleatorio = Math.floor(Math.random() * listaCopia.length); const entrada = listaCopia[indexAleatorio]; const usuarioMinusculas = entrada.user.toLowerCase(); if (blacklist.includes(usuarioMinusculas)) { log.push(`❌ Descartado: ${entrada.ticket} (${entrada.user} está en la Lista Negra)`); listaCopia.splice(indexAleatorio, 1); } else { ganadorEncontrado = entrada; // Posición en la lista original indexGanadorEnListaOriginal = participantesValidos.findIndex(p => p.ticket === entrada.ticket); log.push(`✅ ¡TIRADA GANADORA!: ${entrada.ticket}`); break; } } if (!ganadorEncontrado) { alert("No se pudo encontrar un ganador."); return; } // --- LÓGICA CIRCULAR DE 5 USUARIOS ÚNICOS PARA EL ARCADE --- let anteriores = []; let posteriores = []; let usuariosUsados = new Set(); usuariosUsados.add(ganadorEncontrado.user.toLowerCase()); // 1. Buscar 2 usuarios ÚNICOS anteriores (hacia atrás de forma circular) let curr = indexGanadorEnListaOriginal - 1; let intentosAnteriores = 0; while (anteriores.length < 2 && intentosAnteriores < total) { // Si baja de 0, salta al final de la lista if (curr < 0) { curr = total - 1; } let cand = participantesValidos[curr]; let uLower = cand.user.toLowerCase(); if (!usuariosUsados.has(uLower)) { anteriores.unshift(cand); // Inserta al principio para mantener el orden visual usuariosUsados.add(uLower); } curr--; intentosAnteriores++; } // 2. Buscar 2 usuarios ÚNICOS posteriores (hacia adelante de forma circular) curr = indexGanadorEnListaOriginal + 1; let intentosPosteriores = 0; while (posteriores.length < 2 && intentosPosteriores < total) { // Si supera el final, salta al principio de la lista if (curr >= total) { curr = 0; } let cand = participantesValidos[curr]; let uLower = cand.user.toLowerCase(); if (!usuariosUsados.has(uLower)) { posteriores.push(cand); usuariosUsados.add(uLower); } curr++; intentosPosteriores++; } // Grupo final ordenado: [Anterior 2, Anterior 1, Ganador, Posterior 1, Posterior 2] const arcadeGroup = [...anteriores, ganadorEncontrado, ...posteriores]; if (confirm(`Ganador extraído: ${ganadorEncontrado.user} (${ganadorEncontrado.ticket}). ¿Publicar resultado?`)) { const ahora = new Date(); const fechaFormateada = ahora.toLocaleDateString('es-ES') + ' ' + ahora.toLocaleTimeString('es-ES', {hour: '2-digit', minute:'2-digit'}); db.ref('sorteo_actual').set({ ganador: ganadorEncontrado, arcadeGroup: arcadeGroup, log: log, timestamp: Date.now() }); db.ref('historial_ganadores').push({ user: ganadorEncontrado.user, ticket: ganadorEncontrado.ticket, fecha: fechaFormateada, timestamp: Date.now() }); } }
