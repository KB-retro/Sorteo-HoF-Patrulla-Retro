  <script>
    // Cargar ganadores iniciales por defecto
    document.getElementById('excludedUsers').value = `@maxmad1984

    function parseTickets() {
      const text = document.getElementById('rawList').value;
      const lines = text.split('\n');
      const tickets = [];
      const users = new Set();

      const regex = /(\d+)\s*-\s*@?([a-zA-Z0-9_]+)/;

      lines.forEach(line => {
        const match = line.match(regex);
        if (match) {
          const num = parseInt(match[1]);
          const user = '@' + match[2].trim().toLowerCase();
          tickets.push({ number: num, user: user, rawUser: '@' + match[2].trim() });
          users.add(user);
        }
      });

      document.getElementById('ticketCount').innerText = `${tickets.length} papeletas válidas`;
      document.getElementById('userCount').innerText = `${users.size} usuarios únicos`;

      return tickets;
    }

    function getExcludedUsers() {
      const raw = document.getElementById('excludedUsers').value;
      return raw.split(/[\n,]/)
        .map(u => u.trim().toLowerCase())
        .filter(u => u.length > 0)
        .map(u => u.startsWith('@') ? u : '@' + u);
    }

    document.getElementById('rawList').addEventListener('input', parseTickets);

    async function runDraw() {
      const tickets = parseTickets();
      const excluded = getExcludedUsers();
      const logBox = document.getElementById('retryLog');
      const winnerBox = document.getElementById('winnerBox');
      const btn = document.getElementById('drawBtn');

      if (tickets.length === 0) {
        alert('Por favor, pega una lista de números válida antes de sortear.');
        return;
      }

      btn.disabled = true;
      logBox.innerHTML = '';

      let validWinner = null;
      let attempts = 0;

      while (!validWinner) {
        attempts++;
        // Selección aleatoria
        const randomIndex = Math.floor(Math.random() * tickets.length);
        const drawn = tickets[randomIndex];

        // Animación visual rápida en pantalla
        winnerBox.innerHTML = `
          <div class="number-display">#${drawn.number}</div>
          <div class="user-display">${drawn.rawUser}</div>
          <small style="color:var(--muted)">Comprobando estado...</small>
        `;

        await new Promise(r => setTimeout(r, 400)); // pausa para efecto de giro

        if (excluded.includes(drawn.user)) {
          const entry = document.createElement('div');
          entry.className = 'log-entry log-rejected';
          entry.innerText = `❌ Tirada #${attempts}: Salió el N° ${drawn.number} (${drawn.rawUser}), pero YA HA GANADO PREMIO. Descartando y reintentando...`;
          logBox.appendChild(entry);
          logBox.scrollTop = logBox.scrollHeight;
        } else {
          validWinner = drawn;
          const entry = document.createElement('div');
          entry.className = 'log-entry log-success';
          entry.innerText = `✅ Tirada #${attempts}: Salió el N° ${drawn.number} (${drawn.rawUser}) -> ¡GANADOR VÁLIDO!`;
          logBox.appendChild(entry);
          logBox.scrollTop = logBox.scrollHeight;
        }
      }

      // Mostrar ganador final
      winnerBox.innerHTML = `
        <div style="color:var(--success); font-weight:bold;">🎉 ¡NÚMERO Y GANADOR VÁLIDO! 🎉</div>
        <div class="number-display">N° ${validWinner.number}</div>
        <div class="user-display">${validWinner.rawUser}</div>
      `;

      // Auto-excluir para el siguiente sorteo
      if (document.getElementById('autoExclude').checked) {
        const currentEx = document.getElementById('excludedUsers').value.trim();
        document.getElementById('excludedUsers').value = currentEx + '\n' + validWinner.rawUser;
      }

      btn.disabled = false;
    }
  </script>
